# Python dependencies for The Second-Rate Backup Band
# Install with: pip install -r requirements.txt
#
# System dependencies (install separately):
#   FluidSynth: brew install fluid-synth  /  apt-get install fluidsynth
#   FFmpeg:     brew install ffmpeg        /  apt-get install ffmpeg

# MIDI file generation (all scripts)
mido

# Pipeline orchestration
pyyaml

# Numerical processing (drum_cleaner, synth_cleaner)
numpy
scipy

# Audio analysis and pitch tracking (drum_cleaner, synth_cleaner)
librosa

# Neural net drum transcription (drum_cleaner) — optional but recommended
# If install fails, drum_cleaner.py falls back to librosa automatically
# macOS: brew install libsndfile first
# Linux: apt-get install libsndfile1-dev first
madmom

# Audio format conversion, e.g. MP3 -> WAV (drum_cleaner, synth_cleaner)
pydub
